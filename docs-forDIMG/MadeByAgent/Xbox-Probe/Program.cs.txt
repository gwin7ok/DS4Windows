using System;
using System.Collections.Generic;
using System.Runtime.InteropServices;
using System.Text;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.Win32.SafeHandles;

// 調査用: HidHide で隠した USB の Xbox コントローラーを、このプログラムから読めるかを確かめる。
// (1) XInput の API（4 スロット）、(2) HID の機器（VID_045E／PID_028E）の生のレポート、の両方を試す。
static class Program
{
    [StructLayout(LayoutKind.Sequential)]
    struct XINPUT_GAMEPAD : IEquatable<XINPUT_GAMEPAD> { public bool Equals(XINPUT_GAMEPAD o) => wButtons == o.wButtons && bLeftTrigger == o.bLeftTrigger && bRightTrigger == o.bRightTrigger && sThumbLX == o.sThumbLX && sThumbLY == o.sThumbLY && sThumbRX == o.sThumbRX && sThumbRY == o.sThumbRY; public ushort wButtons; public byte bLeftTrigger, bRightTrigger; public short sThumbLX, sThumbLY, sThumbRX, sThumbRY; }
    [StructLayout(LayoutKind.Sequential)]
    struct XINPUT_STATE { public uint dwPacketNumber; public XINPUT_GAMEPAD Gamepad; }
    [DllImport("xinput1_4.dll")] static extern uint XInputGetState(uint idx, out XINPUT_STATE s);

    static readonly Guid HidGuid = new Guid("4D1E55B2-F16F-11CF-88CB-001111000030");
    [StructLayout(LayoutKind.Sequential)] struct SP_DEVICE_INTERFACE_DATA { public int cbSize; public Guid g; public int flags; public IntPtr r; }
    [DllImport("setupapi.dll", SetLastError = true)] static extern IntPtr SetupDiGetClassDevs(ref Guid g, IntPtr e, IntPtr h, int f);
    [DllImport("setupapi.dll", SetLastError = true)] static extern bool SetupDiEnumDeviceInterfaces(IntPtr s, IntPtr d, ref Guid g, int i, ref SP_DEVICE_INTERFACE_DATA o);
    [DllImport("setupapi.dll", SetLastError = true, CharSet = CharSet.Unicode)] static extern bool SetupDiGetDeviceInterfaceDetail(IntPtr s, ref SP_DEVICE_INTERFACE_DATA d, IntPtr det, int size, out int req, IntPtr info);
    [DllImport("setupapi.dll")] static extern bool SetupDiDestroyDeviceInfoList(IntPtr s);
    [DllImport("kernel32.dll", SetLastError = true, CharSet = CharSet.Unicode)] static extern SafeFileHandle CreateFile(string n, uint acc, uint share, IntPtr sec, uint disp, uint flags, IntPtr t);
    [DllImport("kernel32.dll", SetLastError = true)] static extern bool ReadFile(SafeFileHandle h, byte[] b, int n, out int read, IntPtr ov);

    static List<string> FindHidPaths(string vidPid)
    {
        var list = new List<string>();
        Guid g = HidGuid;
        IntPtr set = SetupDiGetClassDevs(ref g, IntPtr.Zero, IntPtr.Zero, 0x2 | 0x10); // PRESENT | DEVICEINTERFACE
        for (int i = 0; ; i++)
        {
            var d = new SP_DEVICE_INTERFACE_DATA { cbSize = Marshal.SizeOf<SP_DEVICE_INTERFACE_DATA>() };
            if (!SetupDiEnumDeviceInterfaces(set, IntPtr.Zero, ref g, i, ref d)) break;
            SetupDiGetDeviceInterfaceDetail(set, ref d, IntPtr.Zero, 0, out int req, IntPtr.Zero);
            IntPtr buf = Marshal.AllocHGlobal(req);
            Marshal.WriteInt32(buf, IntPtr.Size == 8 ? 8 : 6);
            if (SetupDiGetDeviceInterfaceDetail(set, ref d, buf, req, out _, IntPtr.Zero))
            {
                string path = Marshal.PtrToStringUni(buf + 4);
                if (path.IndexOf(vidPid, StringComparison.OrdinalIgnoreCase) >= 0) list.Add(path);
            }
            Marshal.FreeHGlobal(buf);
        }
        SetupDiDestroyDeviceInfoList(set);
        return list;
    }

    static int Main(string[] args)
    {
        int seconds = args.Length > 0 ? int.Parse(args[0]) : 10;
        Console.OutputEncoding = Encoding.UTF8;
        Console.WriteLine("== XboxProbe: {0} 秒間、ボタンやスティックを操作してください ==", seconds);

        // (2) HID
        var paths = FindHidPaths("vid_045e&pid_028e");
        Console.WriteLine("[HID] VID_045E&PID_028E の機器: {0} 件", paths.Count);
        foreach (string path in paths)
        {
            Console.WriteLine("[HID] path = {0}", path);
            SafeFileHandle h = CreateFile(path, 0x80000000, 3, IntPtr.Zero, 3, 0, IntPtr.Zero);
            if (h.IsInvalid) { Console.WriteLine("[HID]   開けない: Win32 エラー {0}", Marshal.GetLastWin32Error()); continue; }
            Console.WriteLine("[HID]   開けた");
            string p = path;
            var t = new Thread(() =>
            {
                var buf = new byte[64]; byte[] last = null;
                while (true)
                {
                    if (!ReadFile(h, buf, buf.Length, out int n, IntPtr.Zero)) { Console.WriteLine("[HID]   ReadFile 失敗: {0}", Marshal.GetLastWin32Error()); return; }
                    var cur = new byte[n]; Array.Copy(buf, cur, n);
                    if (last == null || !((ReadOnlySpan<byte>)cur).SequenceEqual(last))
                    { Console.WriteLine("[HID]   {0} バイト: {1}", n, BitConverter.ToString(cur)); last = cur; }
                }
            }) { IsBackground = true };
            t.Start();
        }

        // (1) XInput
        var lastPad = new XINPUT_GAMEPAD[4]; var lastKnown = new bool[4]; var seen = new bool[4];
        DateTime end = DateTime.UtcNow.AddSeconds(seconds);
        while (DateTime.UtcNow < end)
        {
            for (uint i = 0; i < 4; i++)
            {
                uint r = XInputGetState(i, out XINPUT_STATE s);
                if (r == 0)
                {
                    if (!seen[i]) { Console.WriteLine("[XInput] スロット {0}: 接続あり", i); seen[i] = true; }
                    if (!lastKnown[i] || !s.Gamepad.Equals(lastPad[i]))
                    {
                        lastPad[i] = s.Gamepad; lastKnown[i] = true;
                        Console.WriteLine("[XInput] スロット {0}: ボタン 0x{1:X4} LT={2} RT={3} LX={4} LY={5} RX={6} RY={7}", i, s.Gamepad.wButtons, s.Gamepad.bLeftTrigger, s.Gamepad.bRightTrigger, s.Gamepad.sThumbLX, s.Gamepad.sThumbLY, s.Gamepad.sThumbRX, s.Gamepad.sThumbRY);
                    }
                }
            }
            Thread.Sleep(20);
        }
        for (int i = 0; i < 4; i++) if (!seen[i]) Console.WriteLine("[XInput] スロット {0}: 接続なし", i);
        Console.WriteLine("== 終了 ==");
        Environment.Exit(0);
        return 0;
    }
}
